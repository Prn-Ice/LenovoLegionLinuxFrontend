function(add_helper target)
  add_executable(${target} helper_bootstrap.cc control_session.cc)
  target_compile_features(${target} PRIVATE cxx_std_14)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(${target} PRIVATE package_power_sampler helper_cli control_dispatch trusted_executable executable_snapshot helper_lifetime)
endfunction()

if(BUILD_CONTROL_HELPER)
  add_helper(legion_control_helper)
  if(BUILD_CONTROL_STATIC)
    target_link_options(legion_control_helper PRIVATE -static -Wl,--gc-sections -Wl,-t)
    add_executable(legion_helper_cleanup helper_cleanup.cc)
    target_compile_features(legion_helper_cleanup PRIVATE cxx_std_14)
    target_compile_options(legion_helper_cleanup PRIVATE
      -Os -fno-pie -fno-stack-protector -fno-exceptions -fno-rtti
      -fno-asynchronous-unwind-tables -fno-unwind-tables)
    target_link_options(legion_helper_cleanup PRIVATE
      -nostdlib -static -no-pie -Wl,--build-id=none -Wl,-e,_start)
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/helper_cleanup_image.h"
      COMMAND ${CMAKE_COMMAND} -DINPUT=$<TARGET_FILE:legion_helper_cleanup>
        -DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/helper_cleanup_image.h
        -P "${CMAKE_CURRENT_SOURCE_DIR}/embed_cleanup.cmake"
      DEPENDS legion_helper_cleanup embed_cleanup.cmake VERBATIM)
    add_custom_target(helper_cleanup_image DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/helper_cleanup_image.h")
    add_dependencies(legion_control_helper helper_cleanup_image)
    target_include_directories(legion_control_helper PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_definitions(legion_control_helper PRIVATE LEGION_EMBEDDED_CLEANUP)
  endif()
  if(CMAKE_VERSION VERSION_LESS 3.16)
    message(FATAL_ERROR "App helper builds require CMake 3.16 or later.")
  endif()
  # Nix wrappers inject development rpaths. Keep only canonical store libraries;
  # other hosts use the loader cache. Never inherit the desktop bundle rpath.
  set(helper_host_library_paths "")
  foreach(directory IN LISTS GIO_LIBRARY_DIRS CMAKE_CXX_IMPLICIT_LINK_DIRECTORIES)
    get_filename_component(canonical_directory "${directory}" REALPATH)
    if(canonical_directory MATCHES "^/nix/store/" AND IS_DIRECTORY "${canonical_directory}")
      list(APPEND helper_host_library_paths "${canonical_directory}")
    endif()
  endforeach()
  list(REMOVE_DUPLICATES helper_host_library_paths)
  set_target_properties(legion_control_helper PROPERTIES
    OUTPUT_NAME "legion-control-helper"
    INSTALL_REMOVE_ENVIRONMENT_RPATH TRUE
    INSTALL_RPATH_USE_LINK_PATH FALSE
    INSTALL_RPATH "${helper_host_library_paths}")
endif()
